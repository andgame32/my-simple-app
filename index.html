export async function api(path, options = {}) { const token = getToken() const response = await fetch(`/api${path}`, { method: options.method || 'GET', headers: { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}), ...(options.headers || {}), }, ...options, }) const data = await response.json().catch(() => ({})) if (!response.ok) { throw new Error(data.message || `HTTP ${response.status}`) } return data }
