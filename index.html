export async function api(path, options = {}) { const token = getToken() const cleanPath = path.startsWith('/') ? path : `/${path}` const response = await fetch(`/api${cleanPath}`, { method: options.method || 'GET', headers: { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}), ...(options.headers || {}), }, body: options.body, }) const text = await response.text() let data try { data = JSON.parse(text) } catch { data = { message: text } } if (!response.ok) { throw new Error( data.message || `HTTP ${response.status}` ) } return data }
