export async function api(path, options = {}) { const token = getToken() const isFormData = options.body instanceof FormData const base = API_URL.replace(/\/$/, '') const cleanPath = path.startsWith('/') ? path : `/${path}` const response = await fetch(`${base}${cleanPath}`, { method: options.method || 'GET', headers: { ...(!isFormData ? { 'Content-Type': 'application/json' } : {}), ...(token ? { Authorization: `Bearer ${token}` } : {}), ...(options.headers || {}), }, ...options, }) let data try { data = await response.json() } catch { data = {} } if (!response.ok) { const error = data.message || Object.values(data.errors || {})?.flat()?.[0] || `HTTP ${response.status}` throw new Error(error) } return data }
